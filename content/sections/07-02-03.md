Un foro in una lastra tesa **concentra** le tensioni: per una lastra infinita la teoria di Kirsch dà `σmax = 3·σ` sul bordo del foro, perpendicolarmente al carico. Con **scikit-fem** (elementi finiti in puro Python) risolvi lo stato piano di tensione e ritrovi il fattore `Kt` numericamente.

```python
import numpy as np
import skfem as fem

m = fem.MeshTri().refined(3)                   # un quadrato unitario diviso in triangoli
print(m)
print("nodi:", m.p.shape[1], "elementi:", m.t.shape[1])
```

Nell'app scikit-fem si scarica da PyPI al primo uso.

## La mesh del quarto di lastra

Per simmetria basta un quarto. La mesh si costruisce a mano in coordinate polari, con elementi più piccoli vicino al foro dove le tensioni cambiano rapidamente.

```python
import numpy as np
import skfem as fem
import matplotlib.pyplot as plt

a, W = 1.0, 6.0                                 # raggio del foro e mezza larghezza (mm)
nr, nt = 16, 20
th = np.linspace(0, np.pi / 2, nt + 1)
ox = np.where(th <= np.pi / 4, W, W / np.tan(np.maximum(th, 1e-12)))   # bordo esterno quadrato
oy = np.where(th <= np.pi / 4, W * np.tan(th), W)
s = np.linspace(0, 1, nr + 1) ** 1.8                                   # infittimento verso il foro
X = a * np.cos(th) + s[:, None] * (ox - a * np.cos(th))
Y = a * np.sin(th) + s[:, None] * (oy - a * np.sin(th))
idx = np.arange(X.size).reshape(X.shape)
t = np.vstack([idx[:-1, :-1].ravel(), idx[1:, :-1].ravel(), idx[1:, 1:].ravel(), idx[:-1, 1:].ravel()])
m = fem.MeshQuad(np.vstack([X.ravel(), Y.ravel()]), t)

fig, ax = plt.subplots(figsize=(4, 4))
for q in m.t.T:
    ax.fill(*m.p[:, q], fc="none", ec="tab:blue", lw=0.4)
ax.set_aspect("equal"); ax.set_title(f"{m.t.shape[1]} elementi quadrilateri")
```

## Soluzione e fattore di concentrazione

Elasticità lineare in stato piano di tensione, elementi quadratici, trazione unitaria sul lato destro, simmetrie sugli assi.

```python
import numpy as np
import skfem as fem
from skfem.models.elasticity import linear_elasticity, lame_parameters

a, W, nr, nt = 1.0, 6.0, 24, 32
th = np.linspace(0, np.pi / 2, nt + 1)
ox = np.where(th <= np.pi / 4, W, W / np.tan(np.maximum(th, 1e-12)))
oy = np.where(th <= np.pi / 4, W * np.tan(th), W)
s = np.linspace(0, 1, nr + 1) ** 1.8
X = a * np.cos(th) + s[:, None] * (ox - a * np.cos(th))
Y = a * np.sin(th) + s[:, None] * (oy - a * np.sin(th))
idx = np.arange(X.size).reshape(X.shape)
t = np.vstack([idx[:-1, :-1].ravel(), idx[1:, :-1].ravel(), idx[1:, 1:].ravel(), idx[:-1, 1:].ravel()])
m = fem.MeshQuad(np.vstack([X.ravel(), Y.ravel()]), t)

e = fem.ElementVector(fem.ElementQuad2())
ib = fem.Basis(m, e, intorder=4)
E, nu = 210e3, 0.3
lam, mu = lame_parameters(E, nu)
lam = 2 * lam * mu / (lam + 2 * mu)            # correzione per lo stato piano di tensione
K = linear_elasticity(lam, mu).assemble(ib)

@fem.LinearForm
def trazione(v, w):
    return 1.0 * v[0]                           # σ = 1 MPa verso destra
fb = fem.FacetBasis(m, e, facets=m.facets_satisfying(lambda x: np.isclose(x[0], W)), intorder=4)
f = trazione.assemble(fb)

D = np.concatenate([
    ib.get_dofs(lambda x: np.isclose(x[0], 0)).all("u^1"),     # simmetria: ux = 0 sull'asse y
    ib.get_dofs(lambda x: np.isclose(x[1], 0)).all("u^2"),     # simmetria: uy = 0 sull'asse x
])
u = fem.solve(*fem.condense(K, f, D=D))

G = ib.interpolate(u).grad                      # gradiente degli spostamenti nei punti di Gauss
sxx = 2 * mu * G[0, 0] + lam * (G[0, 0] + G[1, 1])
sb = fem.Basis(m, fem.ElementQuad2(), intorder=4)
s_nodi = sb.project(sxx)
k = np.argmin(sb.doflocs[0] ** 2 + (sb.doflocs[1] - a) ** 2)  # nodo in cima al foro
print(f"Kt = σmax / σ = {s_nodi[k]:.2f}   (Kirsch, lastra infinita: 3.00)")
```

Il valore è un po' sopra 3 perché la lastra ha larghezza finita (12 raggi): allontanando i bordi si avvicina a 3.

## Mappa delle tensioni

```python
import numpy as np
import skfem as fem
import matplotlib.pyplot as plt
from skfem.models.elasticity import linear_elasticity, lame_parameters

a, W, nr, nt = 1.0, 6.0, 16, 24
th = np.linspace(0, np.pi / 2, nt + 1)
ox = np.where(th <= np.pi / 4, W, W / np.tan(np.maximum(th, 1e-12)))
oy = np.where(th <= np.pi / 4, W * np.tan(th), W)
s = np.linspace(0, 1, nr + 1) ** 1.8
X = a * np.cos(th) + s[:, None] * (ox - a * np.cos(th))
Y = a * np.sin(th) + s[:, None] * (oy - a * np.sin(th))
idx = np.arange(X.size).reshape(X.shape)
t = np.vstack([idx[:-1, :-1].ravel(), idx[1:, :-1].ravel(), idx[1:, 1:].ravel(), idx[:-1, 1:].ravel()])
m = fem.MeshQuad(np.vstack([X.ravel(), Y.ravel()]), t)
e = fem.ElementVector(fem.ElementQuad1())
ib = fem.Basis(m, e)
lam, mu = lame_parameters(210e3, 0.3); lam = 2 * lam * mu / (lam + 2 * mu)
K = linear_elasticity(lam, mu).assemble(ib)
fb = fem.FacetBasis(m, e, facets=m.facets_satisfying(lambda x: np.isclose(x[0], W)))
f = fem.LinearForm(lambda v, w: v[0]).assemble(fb)
D = np.concatenate([ib.get_dofs(lambda x: np.isclose(x[0], 0)).all("u^1"), ib.get_dofs(lambda x: np.isclose(x[1], 0)).all("u^2")])
u = fem.solve(*fem.condense(K, f, D=D))
G = ib.interpolate(u).grad
sxx = (2 * mu * G[0, 0] + lam * (G[0, 0] + G[1, 1])).mean(axis=1)    # un valore per elemento

fig, ax = plt.subplots(figsize=(4.6, 4))
from matplotlib.collections import PolyCollection
pc = PolyCollection([m.p[:, q].T for q in m.t.T], array=sxx, cmap="turbo", edgecolor="none")
ax.add_collection(pc); ax.autoscale(); ax.set_aspect("equal")
fig.colorbar(pc, ax=ax, label="σxx [MPa]")
ax.set_title("σxx con σ = 1 MPa applicata")
```

> [!trappola] Con elementi quadratici ci sono gradi di libertà anche a metà dei lati: per vincolarli tutti serve `.all("u^1")`, non solo i nodi ai vertici. Dimenticarlo dà risultati sbagliati senza nessun errore.

> [!ing] Il `Kt` che ricavi qui è quello delle tabelle di Peterson usate in Costruzione di macchine: ora sai da dove vengono quei grafici, e come calcolarlo per una geometria che nelle tabelle non c'è.
